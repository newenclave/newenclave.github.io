fullaz-db

A typed database composition layer for Zig. Pick a few page-backed data structures, put them in one schema, and update them inside one explicit transaction.

Open the Helsinki dispatch demo →
What it is not

Not SQL, not an ORM, not a production-ready general DBMS. It is a small, explicit building block for durable indexed applications.

Why use it?

One transaction

Change a primary index, a spatial index, a queue, an audit stream, and a document together. Commit or roll back the entire operation.

Typed at compile time

Your schema, page kinds, component options, and durable layout are known before the program starts.

Small primitives

Use the structure that fits: ordered keys, map windows, priority work, append-only bytes, or editable text.

The small API surface

SchemaNames the components in one database.
bptOrdered unique key/value index.
rtree2D or N-dimensional area index.
slotHeapPersistent priority queue.
chainStoreOne durable appendable byte stream.
weightedSequenceOne editable byte document.
MemoryDatabaseFast, ephemeral, browser-friendly backend.
StaticDatabasePersistent image with dirty-image detection.
StaticDatabaseWithWalPersistent image with a small redo WAL.

Start small

1. Create a database

A schema can be empty. This is the smallest useful shape for learning the lifecycle.

const std = @import("std");
const fullaz_db = @import("fullaz-db");

const Schema = fullaz_db.Schema(.{ .page_id = u32 });
const Database = fullaz_db.MemoryDatabase(Schema);

pub fn main() !void {
    var database = try Database.init(std.heap.page_allocator, .{
        .page_size = 4096,
        .cache_frames = 16,
    });
    defer database.deinit();
}

2. Add one ordered index

`bpt` stores unique byte keys in comparator order. This complete example creates an in-memory database from the schema before opening a transaction.

const std = @import("std");
const fullaz = @import("fullaz");
const fullaz_db = @import("fullaz-db");

fn compare(_: void, left: []const u8, right: []const u8) fullaz.core.algorithm.Order {
    return switch (std.mem.order(u8, left, right)) {
        .lt => .lt, .eq => .eq, .gt => .gt,
    };
}

const Schema = fullaz_db.Schema(.{ .page_id = u32 }).add("users", fullaz_db.bpt(.{
    .compare = compare,
    .CompareContext = void,
    .comparator_id = 1,
    .maximum_key_size = 32,
    .maximum_value_size = 128,
}));

const Database = fullaz_db.MemoryDatabase(Schema);

pub fn main() !void {
    var database = try Database.init(std.heap.page_allocator, .{
        .page_size = 4096,
        .cache_frames = 16,
    });
    defer database.deinit();

    var tx = try database.begin();
    defer tx.deinit();
    _ = try tx.get("users").insert("ada", "Ada Lovelace");
    try tx.commit();
}

3. Change several indexes atomically

Add a primary B+ tree, an R-tree for areas, and a priority queue. This Helsinki pump order enters all three indexes in one transaction, or none of them.

const std = @import("std");
const fullaz = @import("fullaz");
const fullaz_db = @import("fullaz-db");

fn compare(_: void, left: []const u8, right: []const u8) fullaz.core.algorithm.Order {
    return switch (std.mem.order(u8, left, right)) {
        .lt => .lt, .eq => .eq, .gt => .gt,
    };
}

fn heapCompare(_: void, left: []const u8, right: []const u8) std.math.Order {
    return std.mem.order(u8, left, right);
}

const Schema = fullaz_db.Schema(.{ .page_id = u32 })
    .add("orders", fullaz_db.bpt(.{
        .compare = compare, .CompareContext = void, .comparator_id = 1,
        .maximum_key_size = 8, .maximum_value_size = 64,
    }))
    .add("areas", fullaz_db.rtree(.{
        .Coord = f32, .dimensions = 2,
        .maximum_entries = 8, .maximum_value_size = 8,
    }))
    .add("queue", fullaz_db.slotHeap(.{
        .compare = heapCompare, .CompareContext = void, .comparator_id = 2,
        .maximum_key_size = 8, .maximum_value_size = 0, .maximum_level = 8,
    }));

const Database = fullaz_db.MemoryDatabase(Schema);

pub fn main() !void {
    var database = try Database.init(std.heap.page_allocator, .{
        .page_size = 4096,
        .cache_frames = 16,
    });
    defer database.deinit();

    const order_id = "20260822"; // sortable due key
    const record = "open|high|north pump";
    const AreaBox = Schema.trait("areas").Binding(Database.BackendType).Proxy.BoundingBox;
    const area = AreaBox.initWith(.{ 60.1650, 24.9300 }, .{ 60.1750, 24.9500 });

    var tx = try database.begin();
    defer tx.deinit();
    _ = try tx.get("orders").insert(order_id, record);
    try tx.get("areas").insert(area, order_id);
    try tx.get("queue").push(order_id, "");
    try tx.commit();
}

4. Roll back deliberately

Nothing is committed until `commit()`. Use `rollback()` for an intentional abort; an active transaction is also rolled back during cleanup.

var tx = try database.begin();
defer tx.deinit();
_ = try tx.get("users").insert("temporary", "will disappear");
try tx.rollback();

6. Persist an image with a WAL

Format creates both the database image and its redo log. Later, open the same pair with the identical schema options; the WAL recovers committed work before the database is exposed.

const std = @import("std");
const fullaz = @import("fullaz");
const fullaz_db = @import("fullaz-db");

const Schema = fullaz_db.Schema(.{ .page_id = u32 })
    .add("notes", fullaz_db.chainStore(.{}));
const Device = fullaz.device.FileBlock(u32);
const Log = fullaz.device.FileLog(u32);
const Database = fullaz_db.StaticDatabaseWithWal(Schema, Device, Log);
const options: Database.InitOptions = .{
    .image_id = [_]u8{0x42} ** 16,
    .components = .{ .notes = .{} },
};

pub fn formatAndWrite(
    allocator: std.mem.Allocator,
    io: std.Io,
    image_path: []const u8,
    wal_path: []const u8,
) !void {
    var database = try Database.format(
        allocator,
        try Device.create(io, image_path, 512),
        try Log.create(io, wal_path),
        options,
    );
    defer database.deinit();

    var tx = try database.begin();
    defer tx.deinit();
    try tx.get("notes").append("Helsinki dispatch started");
    try tx.commit();
}

pub fn reopenAndRead(
    allocator: std.mem.Allocator,
    io: std.Io,
    image_path: []const u8,
    wal_path: []const u8,
) !void {
    var database = try Database.open(
        allocator,
        try Device.open(io, image_path, 512),
        try Log.open(io, wal_path),
        options,
    );
    defer database.deinit();

    var bytes: [64]u8 = undefined;
    const count = try database.getConst("notes").readAt(0, &bytes);
    std.debug.print("{s}\n", .{bytes[0..count]});
}

Persistent CLI

The native dispatch program uses StaticDatabaseWithWal. It keeps an image and a sidecar WAL, and offers a small shell around the same application operations.

zig build run-dispatch -- dispatch.img --format add 20260822 60.1699 24.9384 0.01 open|high|north-pump
zig build run-dispatch -- dispatch.img top
zig build run-dispatch -- dispatch.img area 60.1699 24.9384 0.02
add
add <id> <lat> <lng> <radius> <value...> — insert an order and its indexes.
top
Show the next due order without removing it.
complete
Remove the next due order from every relevant component and append an audit entry.
list
List the orders and their areas in ID order.
area
area <lat> <lng> <radius> — query intersecting service areas.
help
Print all commands. In the interactive shell, quit or exit closes it.

Generate a deterministic global test dataset after building the CLI:

python3 demos/dispatch/tools/generate_locations.py dispatch.img 1000 --format
python3 demos/dispatch/tools/generate_locations.py dispatch.img 1000 --dry-run --seed 42
